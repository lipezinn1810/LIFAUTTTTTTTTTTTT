                                <p>O Integra IF Goiano 2026 ser&aacute; palco de um encontro voltado &agrave; ci&ecirc;ncia, cultura, inova&ccedil;&atilde;o e ao fortalecimento da identidade da Rede Federal. <p>Com o tema <em>&ldquo;Ensino, Pesquisa, Extens&atilde;o e Territ&oacute;rio: O Protagonismo Feminino na Transforma&ccedil;&atilde;o Social&rdquo;</em>, esta edi&ccedil;&atilde;o prop&otilde;e reflex&otilde;es sobre o papel das mulheres na produ&ccedil;&atilde;o do conhecimento, na educa&ccedil;&atilde;o, na ci&ecirc;ncia e nas transforma&ccedil;&otilde;es sociais constru&iacute;das nos diferentes territ&oacute;rios.     $(document).ready(function () {
        var items = $(".palestrante-item"),

            itemsPerPage = 12,
            currentPage = 1;

        function showPage(page) {
            items.hide().slice((page - 1) * itemsPerPage, page * itemsPerPage).show();
        }

        showPage(currentPage);

        function updatePagination() {
            $("#pagination-container").empty();
            var totalPages = Math.ceil(items.length / itemsPerPage),
                maxButtons = 6,
                startPage,
                endPage;

            if (totalPages <= maxButtons) {
                startPage = 1;
                endPage = totalPages;
            } else {
                if (currentPage <= Math.ceil(maxButtons / 2)) {
                    startPage = 1;
                    endPage = maxButtons;
                } else if (currentPage + Math.floor(maxButtons / 2) >= totalPages) {
                    startPage = totalPages - maxButtons + 1;
                    endPage = totalPages;
                } else {
                    startPage = currentPage - Math.floor(maxButtons / 2);
                    endPage = currentPage + Math.floor(maxButtons / 2);
                }
            }

            if (currentPage > 1) {
                $("<button>")
                    .text("<")
                    .addClass("btn btn-outline-success m-1 px-3 prev-btn br-2")
                    .on("click", function () {
                        currentPage--;
                        showPage(currentPage);
                        updatePagination();
                    })
                    .prependTo("#pagination-container");
            }

            if (endPage > 1) {
                for (var i = startPage; i <= endPage; i++) {
                    $("<button>")
                        .text(i)
                        .addClass("btn btn-outline-success m-1 px-3 page-btn br-2")
                        .toggleClass("active", i === currentPage)
                        .on("click", function () {
                            currentPage = parseInt($(this).text());
                            showPage(currentPage);
                            updatePagination();
                        })
                        .appendTo("#pagination-container");
                }
            }

            if (currentPage < totalPages) {
                $("<button>")
                    .text(">")
                    .addClass("btn btn-outline-success m-1 px-3 br-2")
                    .on("click", function () {
                        currentPage++;
                        showPage(currentPage);
                        updatePagination();
                    })
                    .appendTo("#pagination-container");
            }
        }
        updatePagination();
    });
</script>
